// ============================================================
//                    SETUP
// ============================================================
void setup() {
    Serial.begin(115200);
    delay(1000);
    
    Serial.println("========================================");
    Serial.println("ESP32 - NODE 2 (RECEIVER)");
    Serial.println("WITH LCD, AI, WiFi, MQTT, WEB & DIGITAL TWIN");
    Serial.println("CAN TRANSMIT TO NODEMCU ENABLED");
    Serial.println("========================================");
    
    // ---- Initialize Buzzer ----
    pinMode(BUZZER_PIN, OUTPUT);
    digitalWrite(BUZZER_PIN, LOW);
    
    // ---- Initialize LCD ----
    Wire.begin(I2C_SDA, I2C_SCL);
    lcd.init();
    lcd.backlight();
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("CAN Receiver");
    lcd.setCursor(0, 1);
    lcd.print("Node 2 - AI Ready");
    delay(2000);
    lcd.clear();
    
    // ---- Initialize Encoder ----
    pinMode(ENC_CLK, INPUT_PULLUP);
    pinMode(ENC_DT, INPUT_PULLUP);
    pinMode(ENC_SW, INPUT_PULLUP);
    
    // ---- Initialize WiFi ----
    lcd.setCursor(0, 0);
    lcd.print("Connecting WiFi");
    lcd.setCursor(0, 1);
    lcd.print("Please wait...");
    
    WiFi.begin(ssid, password);
    int attempts = 0;
    while (WiFi.status() != WL_CONNECTED && attempts < 20) {
        delay(500);
        Serial.print(".");
        attempts++;
    }
    
    if (WiFi.status() == WL_CONNECTED) {
        wifiConnected = true;
        Serial.println("\n✅ WiFi Connected!");
        Serial.print("IP Address: ");
        Serial.println(WiFi.localIP());
        
        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("WiFi Connected!");
        lcd.setCursor(0, 1);
        lcd.print(WiFi.localIP().toString());
        delay(2000);
    } else {
        Serial.println("\n❌ WiFi Connection Failed!");
        wifiConnected = false;
        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("WiFi Failed!");
        lcd.setCursor(0, 1);
        lcd.print("Check Credentials");
        delay(2000);
    }
    
    // ---- Initialize MQTT ----
    if (wifiConnected) {
        mqttClient.setServer(mqtt_server, mqtt_port);
        mqttClient.setCallback(mqttCallback);
        connectMQTT();
    }
    
    // ---- Initialize Web Server with Digital Twin ----
    if (wifiConnected) {
        setupWebServer();
        server.begin();
        Serial.println("✅ Web Server Started!");
        Serial.print("Web Dashboard: http://");
        Serial.println(WiFi.localIP());
        Serial.print("Digital Twin: http://");
        Serial.print(WiFi.localIP());
        Serial.println("/twin");
    }
    
    // ---- Initialize CAN (Receiver) ----
    CAN.setPins(RX_GPIO_NUM, TX_GPIO_NUM);
    
    if (!CAN.begin(250E3)) {
        Serial.println("❌ Starting CAN failed!");
        canStatus = false;
        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("CAN Failed!");
        while (1);
    }
    
    Serial.println("✅ CAN Ready at 250KBPS!");
    canStatus = true;
    
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("CAN Connected!");
    lcd.setCursor(0, 1);
    lcd.print("AI Monitoring ON");
    delay(1000);
    lcd.clear();
    
    Serial.println("Waiting for data...\n");
    Serial.println("AI Predictions Engine: READY");
    Serial.println("Web Dashboard: READY");
    Serial.println("Digital Twin: READY");
    Serial.println("MQTT: READY");
    Serial.println("CAN Transmit to NodeMCU: ENABLED");
    Serial.println("========================================\n");
}

// ============================================================
//                    MAIN LOOP
// ============================================================
void loop() {
    // Read encoder
    readEncoder();
    
    // Receive CAN data
    canReceiver();
    
    // Send heartbeat every 3 seconds
    static unsigned long lastSend = 0;
    if (millis() - lastSend > 3000) {
        sendHeartbeat();
        lastSend = millis();
    }
    
    // Run AI Predictions every 2 seconds
    if (millis() - lastAlertTime > 2000) {
        runAIPredictions();
        lastAlertTime = millis();
    }
    
    // Store historical data every 5 seconds
    if (millis() - lastHistoryUpdate > 5000) {
        storeHistoricalData();
        lastHistoryUpdate = millis();
    }
    
    // Update buzzer
    updateBuzzer();
    
    // ---- TRANSMIT PROCESSED DATA TO NODEMCU VIA CAN ----
    if (millis() - lastCANTransmit > canTransmitInterval) {
        transmitToNodeMCU();
        lastCANTransmit = millis();
    }
    
    // MQTT reconnect if needed
    if (wifiConnected && !mqttClient.connected()) {
        connectMQTT();
    }
    mqttClient.loop();
    
    // Upload to cloud every 10 seconds
    if (wifiConnected && millis() - lastCloudUpload > 10000) {
        uploadToCloud();
        lastCloudUpload = millis();
    }
    
    // Handle web server
    if (wifiConnected) {
        server.handleClient();
    }
    
    // Print full vehicle data every 2 seconds (Serial Monitor)
    if (millis() - lastPrintTime > 2000) {
        printVehicleData();
        printAIData();
        lastPrintTime = millis();
    }
    
    // Update LCD display every 500ms
    if (millis() - lastLCDUpdate > 500) {
        updateLCD();
        lastLCDUpdate = millis();
    }
    
    delay(5);
}

// ============================================================
//                    TRANSMIT TO NODEMCU VIA CAN
// ============================================================
void transmitToNodeMCU() {
    static int msgCounter = 0;
    msgCounter++;
    
    // ---- Send Engine Data (ID: 0x101) ----
    CAN.beginPacket(0x101);
    CAN.write(NODE_ID);                    // From Node 2 (ESP32)
    CAN.write(msgCounter & 0xFF);          // Message counter
    CAN.write((int)(vehicle.rpm) >> 8);    // RPM High byte
    CAN.write((int)(vehicle.rpm) & 0xFF);  // RPM Low byte
    CAN.write((int)(vehicle.speed));       // Speed
    CAN.write((int)(vehicle.engineLoad * 100)); // Engine Load
    CAN.write(vehicle.gear);               // Gear
    CAN.endPacket();
    
    // ---- Send Temperatures (ID: 0x102) ----
    CAN.beginPacket(0x102);
    CAN.write(NODE_ID);
    CAN.write(msgCounter & 0xFF);
    CAN.write((int)(vehicle.coolantTemp));
    CAN.write((int)(vehicle.oilTemp));
    CAN.write((int)(vehicle.exhaustTemp) >> 8);
    CAN.write((int)(vehicle.exhaustTemp) & 0xFF);
    CAN.write((int)(vehicle.catTemp) >> 8);
    CAN.write((int)(vehicle.catTemp) & 0xFF);
    CAN.endPacket();
    
    // ---- Send Performance (ID: 0x103) ----
    CAN.beginPacket(0x103);
    CAN.write(NODE_ID);
    CAN.write(msgCounter & 0xFF);
    CAN.write((int)(vehicle.torque) >> 8);
    CAN.write((int)(vehicle.torque) & 0xFF);
    CAN.write((int)(vehicle.horsepower) >> 8);
    CAN.write((int)(vehicle.horsepower) & 0xFF);
    CAN.write((int)(vehicle.engineHealth));
    CAN.endPacket();
    
    // ---- Send Controls (ID: 0x104) ----
    CAN.beginPacket(0x104);
    CAN.write(NODE_ID);
    CAN.write(msgCounter & 0xFF);
    CAN.write(vehicle.throttlePct);
    CAN.write(vehicle.brakePct);
    CAN.write(vehicle.steeringDeg + 90);
    CAN.write((int)(vehicle.ignitionTiming * 10));
    CAN.write((int)(vehicle.fuelInjectorPulse));
    CAN.endPacket();
    
    // ---- Send Electrical (ID: 0x109) ----
    CAN.beginPacket(0x109);
    CAN.write(NODE_ID);
    CAN.write(msgCounter & 0xFF);
    CAN.write((int)(vehicle.batteryVoltage * 10));
    CAN.write((int)(vehicle.alternatorVoltage * 10));
    CAN.write((int)(vehicle.batterySOC));
    CAN.write((int)(vehicle.batteryHealth));
    CAN.endPacket();
    
    // ---- Send Fuel (ID: 0x10A) ----
    CAN.beginPacket(0x10A);
    CAN.write(NODE_ID);
    CAN.write(msgCounter & 0xFF);
    CAN.write((int)(vehicle.fuelLevel * 10));
    CAN.write((int)(vehicle.fuelConsumption * 10));
    CAN.write((int)(vehicle.fuelPressure * 10));
    CAN.write((int)(vehicle.distance * 10));
    CAN.write((int)(vehicle.maxSpeed));
    CAN.endPacket();
    
    // ---- Send Tyre Pressures (ID: 0x106) ----
    CAN.beginPacket(0x106);
    CAN.write(NODE_ID);
    CAN.write(msgCounter & 0xFF);
    CAN.write((int)(vehicle.tyrePressureFL * 10));
    CAN.write((int)(vehicle.tyrePressureFR * 10));
    CAN.write((int)(vehicle.tyrePressureRL * 10));
    CAN.write((int)(vehicle.tyrePressureRR * 10));
    CAN.endPacket();
    
    // ---- Send Diagnostics (ID: 0x10C) ----
    CAN.beginPacket(0x10C);
    CAN.write(NODE_ID);
    CAN.write(msgCounter & 0xFF);
    byte diagFlags = 0;
    if (vehicle.checkEngineLight) diagFlags |= 0x01;
    if (vehicle.tpmsWarning) diagFlags |= 0x02;
    if (vehicle.serviceRequired) diagFlags |= 0x04;
    if (vehicle.absWarning) diagFlags |= 0x08;
    if (vehicle.brakeWarning) diagFlags |= 0x10;
    if (vehicle.oilWarning) diagFlags |= 0x20;
    if (vehicle.coolantWarning) diagFlags |= 0x40;
    if (vehicle.batteryWarning) diagFlags |= 0x80;
    
    byte diagFlags2 = 0;
    if (vehicle.tractionControlActive) diagFlags2 |= 0x01;
    if (vehicle.espActive) diagFlags2 |= 0x02;
    if (vehicle.engineRunning) diagFlags2 |= 0x04;
    if (vehicle.headlightOn) diagFlags2 |= 0x08;
    if (vehicle.hornOn) diagFlags2 |= 0x10;
    if (vehicle.wiperOn) diagFlags2 |= 0x20;
    
    CAN.write(diagFlags);
    CAN.write(diagFlags2);
    CAN.endPacket();
    
    // ---- Send Mode/Status (ID: 0x111) ----
    CAN.beginPacket(0x111);
    CAN.write(NODE_ID);
    CAN.write(msgCounter & 0xFF);
    CAN.write(vehicle.modeIndex);
    CAN.write(vehicle.pressCount);
    CAN.write(vehicle.gear);
    CAN.write((int)(vehicle.driveShaftSpeed));
    CAN.write((int)(vehicle.engineHours));
    CAN.endPacket();
    
    // ---- Send AI Health Data (ID: 0x115) ----
    CAN.beginPacket(0x115);
    CAN.write(NODE_ID);
    CAN.write(msgCounter & 0xFF);
    CAN.write((int)(aiData.engineHealth));
    CAN.write((int)(aiData.batteryHealth));
    CAN.write((int)(aiData.brakeHealth));
    CAN.write((int)(aiData.oilHealth));
    CAN.write((int)(aiData.overallHealth));
    CAN.write(aiData.alertLevel);
    CAN.endPacket();
    // Add these to transmitToNodeMCU() function:

// ---- Send Brake Data (ID: 0x105) ----
CAN.beginPacket(0x105);
CAN.write(NODE_ID);
CAN.write(msgCounter & 0xFF);
CAN.write((int)(vehicle.brakePadWear));
CAN.write((int)(vehicle.brakeDiscWear));
CAN.write((int)(vehicle.brakeTemp));
CAN.write((int)(vehicle.brakePressure));
CAN.write((int)(vehicle.brakeFluidLevel));
CAN.endPacket();

// ---- Send Tyre Temps (ID: 0x107) ----
CAN.beginPacket(0x107);
CAN.write(NODE_ID);
CAN.write(msgCounter & 0xFF);
CAN.write((int)(vehicle.tyreTempFL));

CAN.write((int)(vehicle.tyreTempRL));
CAN.write((int)(vehicle.tyreTempRR));
CAN.endPacket();

CAN.write((int)(vehicle.tyreWearFL));
CAN.write((int)(vehicle.tyreWearFR));
CAN.write((int)(vehicle.tyreWearRL));
CAN.write((int)(vehicle.tyreWearRR));
CAN.endPacket();

// ---- Send Fluids (ID: 0x10B) ----
CAN.beginPacket(0x10B);
CAN.write(NODE_ID);
CAN.write(msgCounter & 0xFF);
CAN.write((int)(vehicle.oilLevel));
CAN.write((int)(vehicle.oilQuality));
CAN.write((int)(vehicle.oilPressure));
CAN.write((int)(vehicle.coolantLevel));
CAN.write((int)(vehicle.washerFluid));
CAN.endPacket();

// ---- Send MPU6050 Accel (ID: 0x10D) ----
CAN.beginPacket(0x10D);
CAN.write(NODE_ID);
CAN.write(msgCounter & 0xFF);
CAN.write(vehicle.ax >> 8);
CAN.write(vehicle.ax & 0xFF);
CAN.write(vehicle.ay >> 8);
CAN.write(vehicle.ay & 0xFF);
CAN.write(vehicle.az >> 8);
CAN.write(vehicle.az & 0xFF);
CAN.endPacket();

// ---- Send MPU6050 Gyro (ID: 0x10E) ----
CAN.beginPacket(0x10E);
CAN.write(NODE_ID);
CAN.write(msgCounter & 0xFF);
CAN.write(vehicle.gx >> 8);
CAN.write(vehicle.gx & 0xFF);
CAN.write(vehicle.gy >> 8);
CAN.write(vehicle.gy & 0xFF);
CAN.write(vehicle.gz >> 8);
CAN.write(vehicle.gz & 0xFF);
CAN.endPacket();

// ---- Send GPS (ID: 0x10F) ----
int32_t lat = vehicle.latitude * 1000000;
int32_t lng = vehicle.longitude * 1000000;
CAN.beginPacket(0x10F);
CAN.write(NODE_ID);
CAN.write(msgCounter & 0xFF);
CAN.write((lat >> 24) & 0xFF);
CAN.write((lat >> 16) & 0xFF);
CAN.write((lat >> 8) & 0xFF);
CAN.write(lat & 0xFF);
CAN.write((lng >> 24) & 0xFF);
CAN.write((lng >> 16) & 0xFF);
CAN.endPacket();

// ---- Send GPS Altitude (ID: 0x110) ----
CAN.beginPacket(0x110);
CAN.write(NODE_ID);
CAN.write(msgCounter & 0xFF);
CAN.write((int)(vehicle.altitude * 10));
CAN.write(vehicle.satellites);
CAN.write((int)(vehicle.avgSpeed * 10));
CAN.write((int)(vehicle.tripDistance * 10));
CAN.write((int)(vehicle.tripTime * 100));
CAN.endPacket();

// ---- Send Clutch/Transmission (ID: 0x112) ----
CAN.beginPacket(0x112);
CAN.write(NODE_ID);
CAN.write(msgCounter & 0xFF);
CAN.write((int)(vehicle.clutchWear));
CAN.write((int)(vehicle.transmissionHealth));
CAN.endPacket();

// ---- Send Cabin Controls (ID: 0x113) ----
CAN.beginPacket(0x113);
CAN.write(NODE_ID);
CAN.write(msgCounter & 0xFF);
CAN.write(vehicle.acOn ? 1 : 0);
CAN.write(vehicle.heaterOn ? 1 : 0);
CAN.write((int)(vehicle.cabinTemp));
CAN.write((int)(vehicle.fanSpeed * 10));
CAN.endPacket();
    
    // ---- Send Alert Data (ID: 0x116) ----
    CAN.beginPacket(0x116);
    CAN.write(NODE_ID);
    CAN.write(msgCounter & 0xFF);
    CAN.write(aiData.anomalyDetected ? 1 : 0);
    // Send alert message (max 6 chars)
    String msg = aiData.alertMessage.substring(0, 6);
    for (int i = 0; i < 6; i++) {
        if (i < msg.length()) {
            CAN.write(msg.charAt(i));
        } else {
            CAN.write(' ');
        }
    }
    CAN.endPacket();
}

// ============================================================
//                    ENCODER FUNCTIONS (UNCHANGED)
// ============================================================
void readEncoder() {
    int clkState = digitalRead(ENC_CLK);
    
    if (clkState != lastEncCLK) {
        if (digitalRead(ENC_DT) != clkState) {
            if (currentPage < TOTAL_PAGES - 1) {
                currentPage++;
                encoderPos++;
                Serial.print("🔘 Page: ");
                Serial.println(currentPage + 1);
                lcd.clear();
            }
        } else {
            if (currentPage > 0) {
                currentPage--;
                encoderPos--;
                Serial.print("🔘 Page: ");
                Serial.println(currentPage + 1);
                lcd.clear();
            }
        }
        lastEncCLK = clkState;
    }
    
    bool buttonState = digitalRead(ENC_SW);
    if (buttonState == LOW && lastEncoderButton == HIGH) {
        if (millis() - lastEncoderButtonTime > 300) {
            encoderButtonPressed = true;
            lastEncoderButtonTime = millis();
            Serial.println("🔘 Encoder Button Pressed!");
            
            if (currentPage != 0) {
                currentPage = 0;
                encoderPos = 0;
                lcd.clear();
                Serial.println("🔘 Returning to Page 1");
            } else {
                static bool backlightOn = true;
                backlightOn = !backlightOn;
                if (backlightOn) {
                    lcd.backlight();
                } else {
                    lcd.noBacklight();
                }
            }
        }
    }
    lastEncoderButton = buttonState;
}

void connectMQTT() {
    if (!wifiConnected) return;
    
    Serial.print("Connecting to MQTT...");
    if (mqttClient.connect("ESP32_Node2", "", "")) {
        mqttConnected = true;
        Serial.println(" ✅ Connected!");
        mqttClient.subscribe("vehicle/commands");
        lcd.setCursor(0, 1);
        lcd.print("MQTT Connected ");
    } else {
        mqttConnected = false;
        Serial.println(" ❌ Failed!");
    }
}

void mqttCallback(char* topic, byte* payload, unsigned int length) {
    String message;
    for (int i = 0; i < length; i++) {
        message += (char)payload[i];
    }
    
    Serial.print("📨 MQTT Command: ");
    Serial.print(topic);
    Serial.print(" -> ");
    Serial.println(message);
    
    if (String(topic) == "vehicle/commands") {
        if (message == "STATUS") {
            uploadToCloud();
        } else if (message == "HEALTH") {
            sendAIHealthMQTT();
        }
    }
}

// ============================================================
//                    CAN RECEIVER (UNCHANGED)
// ============================================================
void canReceiver() {
    int packetSize = CAN.parsePacket();
    
    if (packetSize) {
        receiveCount++;
        
        byte data[8];
        int i = 0;
        while (CAN.available() && i < 8) {
            data[i++] = CAN.read();
        }
        
        Serial.print("📥 #");
        Serial.print(receiveCount);
        Serial.print(" | ID:0x");
        Serial.print(CAN.packetId(), HEX);
        
        if (i >= 3) {
            Serial.print(" | From:");
            Serial.print(data[0]);
            Serial.print(" | Msg#:");
            Serial.print(data[1]);
            Serial.print(" | Time:");
            Serial.print(data[2]);
            
            if (i > 3) {
                Serial.print(" | HEX:");
                for(int j=3; j<i; j++) {
                    if (data[j] < 0x10) Serial.print("0");
                    Serial.print(data[j], HEX);
                    Serial.print(" ");
                }
            }
        }
        Serial.println();
        
        processCANData(CAN.packetId(), data, i);
    }
}

// ============================================================
//                    PROCESS CAN DATA (UNCHANGED)
// ============================================================
void processCANData(uint32_t packetId, byte data[], int len) {
    if (len < 3) return;
    
    vehicle.nodeId = data[0];
    vehicle.messageCount = data[1];
    vehicle.lastUpdate = millis();
    
    switch(packetId) {
        case 0x101: {
            vehicle.rpm = (data[3] << 8) | data[4];
            vehicle.speed = data[5];
            vehicle.engineLoad = data[6] / 100.0;
            vehicle.gear = data[7];
            break;
        }
            
        case 0x102: {
            vehicle.coolantTemp = data[3];
            vehicle.oilTemp = data[4];
            vehicle.exhaustTemp = (data[5] << 8) | data[6];
            vehicle.catTemp = (data[7] << 8) | data[8];
            break;
        }
            
        case 0x103: {
            vehicle.torque = (data[3] << 8) | data[4];
            vehicle.horsepower = (data[5] << 8) | data[6];
            vehicle.engineHealth = data[7];
            vehicle.engineHours = data[8] / 10.0;
            break;
        }
            
        case 0x104: {
            vehicle.throttlePct = data[3];
            vehicle.brakePct = data[4];
            vehicle.steeringDeg = data[5] - 90;
            vehicle.ignitionTiming = data[6] / 10.0;
            vehicle.fuelInjectorPulse = data[7];
            break;
        }
            
        case 0x105: {
            vehicle.brakePadWear = data[3];
            vehicle.brakeDiscWear = data[4];
            vehicle.brakeTemp = data[5];
            vehicle.brakePressure = data[6];
            vehicle.brakeFluidLevel = data[7];
            break;
        }
            
        case 0x106: {
            vehicle.tyrePressureFL = data[3] / 10.0;
            vehicle.tyrePressureFR = data[4] / 10.0;
            vehicle.tyrePressureRL = data[5] / 10.0;
            vehicle.tyrePressureRR = data[6] / 10.0;
            break;
        }
            
        case 0x107: {
            vehicle.tyreTempFL = data[3];
            vehicle.tyreTempFR = data[4];
            vehicle.tyreTempRL = data[5];
            vehicle.tyreTempRR = data[6];
            break;
        }
            
        case 0x108: {
            vehicle.tyreWearFL = data[3];
            vehicle.tyreWearFR = data[4];
            vehicle.tyreWearRL = data[5];
            vehicle.tyreWearRR = data[6];
            break;
        }
            
        case 0x109: {
            vehicle.batteryVoltage = data[3] / 10.0;
            vehicle.alternatorVoltage = data[4] / 10.0;
            vehicle.batterySOC = data[5];
            vehicle.batteryHealth = data[6];
            vehicle.currentDraw = data[7] * 10.0;
            break;
        }
            
        case 0x10A: {
            vehicle.fuelLevel = data[3] / 10.0;
            vehicle.fuelConsumption = data[4] / 10.0;
            vehicle.fuelPressure = data[5] / 10.0;
            vehicle.distance = data[6] / 10.0;
            vehicle.maxSpeed = data[7];
            break;
        }
            
        case 0x10B: {
            vehicle.oilLevel = data[3];
            vehicle.oilQuality = data[4];
            vehicle.oilPressure = data[5];
            vehicle.coolantLevel = data[6];
            vehicle.washerFluid = data[7];
            break;
        }
            
        case 0x10C: {
            byte diagFlags = data[3];
            byte diagFlags2 = data[4];
            
            vehicle.checkEngineLight = (diagFlags & 0x01) ? true : false;
            vehicle.tpmsWarning = (diagFlags & 0x02) ? true : false;
            vehicle.serviceRequired = (diagFlags & 0x04) ? true : false;
            vehicle.absWarning = (diagFlags & 0x08) ? true : false;
            vehicle.brakeWarning = (diagFlags & 0x10) ? true : false;
            vehicle.oilWarning = (diagFlags & 0x20) ? true : false;
            vehicle.coolantWarning = (diagFlags & 0x40) ? true : false;
            vehicle.batteryWarning = (diagFlags & 0x80) ? true : false;
            
            vehicle.tractionControlActive = (diagFlags2 & 0x01) ? true : false;
            vehicle.espActive = (diagFlags2 & 0x02) ? true : false;
            vehicle.engineRunning = (diagFlags2 & 0x04) ? true : false;
            vehicle.headlightOn = (diagFlags2 & 0x08) ? true : false;
            vehicle.hornOn = (diagFlags2 & 0x10) ? true : false;
            vehicle.wiperOn = (diagFlags2 & 0x20) ? true : false;
            break;
        }
            
        case 0x10D: {
            vehicle.ax = (data[3] << 8) | data[4];
            vehicle.ay = (data[5] << 8) | data[6];
            vehicle.az = (data[7] << 8) | data[8];
            break;
        }
            
        case 0x10E: {
            vehicle.gx = (data[3] << 8) | data[4];
            vehicle.gy = (data[5] << 8) | data[6];
            vehicle.gz = (data[7] << 8) | data[8];
            break;
        }
            
        case 0x10F: {
            int32_t lat = ((uint32_t)data[3] << 24) | ((uint32_t)data[4] << 16) | ((uint32_t)data[5] << 8) | data[6];
            int32_t lng = ((uint32_t)data[7] << 24) | ((uint32_t)data[8] << 16);
            vehicle.latitude = lat / 1000000.0;
            vehicle.longitude = lng / 1000000.0;
            break;
        }
            
        case 0x110: {
            vehicle.altitude = data[3] / 10.0;
            vehicle.satellites = data[4];
            vehicle.avgSpeed = data[5] / 10.0;
            vehicle.tripDistance = data[6] / 10.0;
            vehicle.tripTime = data[7] / 100.0;
            break;
        }
            
        case 0x111: {
            vehicle.modeIndex = data[3];
            vehicle.pressCount = data[4];
            vehicle.gear = data[5];
            vehicle.driveShaftSpeed = data[6];
            vehicle.engineHours = data[7];
            break;
        }
            
        case 0x112: {
            vehicle.clutchWear = data[3];
            vehicle.transmissionHealth = data[4];
            break;
        }
            
        case 0x113: {
            vehicle.acOn = data[3] ? true : false;
            vehicle.heaterOn = data[4] ? true : false;
            vehicle.cabinTemp = data[5];
            vehicle.fanSpeed = data[6] / 10.0;
            break;
        }
            
        case 0x114: {
            if (data[3] == 0xAA && data[4] == 0x55) {
                // Heartbeat received
            }
            break;
        }
    }
}

}

// ============================================================
//                    AI PREDICTIONS ENGINE (UNCHANGED)
// ============================================================
void runAIPredictions() {
    // ---- ENGINE HEALTH PREDICTION ----
    float engineHealth = 100.0;
    
    if (vehicle.rpm > 6000) {
        engineHealth -= (vehicle.rpm - 6000) * 0.02;
    }
    if (vehicle.rpm > 7000) {
        engineHealth -= (vehicle.rpm - 7000) * 0.05;
    }
    
    if (vehicle.coolantTemp > 105) {
        engineHealth -= (vehicle.coolantTemp - 105) * 3;
    }
    if (vehicle.coolantTemp > 115) {
        engineHealth -= (vehicle.coolantTemp - 115) * 5;
    }
    
    if (vehicle.speed > 180) {
        engineHealth -= (vehicle.speed - 180) * 0.5;
    }
    
    if (vehicle.rpm > 4000 && vehicle.speed < 20 && vehicle.speed > 0) {
        engineHealth -= 5;
    }
    
    if (vehicle.clutchWear < 50) {
        engineHealth -= (50 - vehicle.clutchWear) * 0.2;
    }
    
    if (vehicle.oilQuality < 50) {
        engineHealth -= (50 - vehicle.oilQuality) * 0.3;
    }
    
    engineHealth = constrain(engineHealth, 0, 100);
    aiData.engineHealth = engineHealth;
    
    // ---- BATTERY HEALTH PREDICTION ----
    float batteryHealth = vehicle.batteryHealth;
    
    if (vehicle.batteryVoltage < 12.0 && vehicle.batteryVoltage > 0) {
        batteryHealth -= (12.0 - vehicle.batteryVoltage) * 5;
    }
    if (vehicle.batteryVoltage < 11.0) {
        batteryHealth -= (11.0 - vehicle.batteryVoltage) * 10;
    }
    
    if (vehicle.alternatorVoltage < 13.5 && vehicle.rpm > 1000 && vehicle.alternatorVoltage > 0) {
        batteryHealth -= 10;
    }
    if (vehicle.alternatorVoltage > 15.0) {
        batteryHealth -= 5;
    }
    
    if (vehicle.rpm > 5000) {
        batteryHealth -= (vehicle.rpm - 5000) * 0.01;
    }
    
    batteryHealth = constrain(batteryHealth, 0, 100);
    aiData.batteryHealth = batteryHealth;
    
    // ---- BRAKE HEALTH PREDICTION ----
    float brakeHealth = vehicle.brakePadWear;
    
    if (vehicle.brakePct > 80) {
        brakeHealth -= 0.5;
    }
    if (vehicle.brakePct > 95) {
        brakeHealth -= 1.0;
    }
    
    if (vehicle.brakeFluidLevel < 30 && vehicle.brakeFluidLevel > 0) {
        brakeHealth -= (30 - vehicle.brakeFluidLevel) * 0.3;
    }
    
    brakeHealth = constrain(brakeHealth, 0, 100);
    aiData.brakeHealth = brakeHealth;
    
    // ---- OIL HEALTH PREDICTION ----
    float oilHealth = vehicle.oilQuality;
    
    if (vehicle.rpm > 4000) {
        oilHealth -= (vehicle.rpm - 4000) * 0.005;
    }
    
    if (vehicle.coolantTemp > 100) {
        oilHealth -= (vehicle.coolantTemp - 100) * 0.5;
    }
    
    if (vehicle.oilLevel < 50 && vehicle.oilLevel > 0) {
        oilHealth -= (50 - vehicle.oilLevel) * 0.5;
    }
    
    oilHealth = constrain(oilHealth, 0, 100);
    aiData.oilHealth = oilHealth;
    
    // ---- OVERALL HEALTH SCORE ----
    float overallHealth = (engineHealth * 0.40) + 
                          (batteryHealth * 0.25) + 
                          (brakeHealth * 0.20) + 
                          (oilHealth * 0.15);
    aiData.overallHealth = overallHealth;
    
    // ---- ANOMALY DETECTION & ALERT SYSTEM ----
    aiData.anomalyDetected = false;
    aiData.alertMessage = "";
    aiData.alertLevel = 0;
    aiData.alertActive = false;
    
    // Check 1: Engine Overheating
    if (vehicle.coolantTemp > 115 && vehicle.coolantTemp > 0) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "ENGINE OVERHEATING!";
        aiData.alertLevel = 2;
        aiData.alertActive = true;
    }
    else if (vehicle.coolantTemp > 105) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "ENGINE TEMP HIGH";
        aiData.alertLevel = 1;
        aiData.alertActive = true;
    }
    
    // Check 2: High RPM
    else if (vehicle.rpm > 6000) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "HIGH RPM DETECTED!";
        aiData.alertLevel = 2;
        aiData.alertActive = true;
    }
    else if (vehicle.rpm > 5000) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "RPM ABOVE NORMAL";
        aiData.alertLevel = 1;
        aiData.alertActive = true;
    }
    
    // Check 3: Engine Health Critical
    else if (engineHealth < 30) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "ENGINE HEALTH CRITICAL!";
        aiData.alertLevel = 2;
        aiData.alertActive = true;
    }
    else if (engineHealth < 50) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "ENGINE HEALTH LOW";
        aiData.alertLevel = 1;
        aiData.alertActive = true;
    }
    
    // Check 4: Battery Health Critical
    else if (batteryHealth < 30) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "BATTERY HEALTH CRITICAL!";
        aiData.alertLevel = 2;
        aiData.alertActive = true;
    }
    else if (batteryHealth < 50) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "BATTERY HEALTH LOW";
        aiData.alertLevel = 1;
        aiData.alertActive = true;
    }
    
    // Check 5: Brake Pad Wear
    else if (vehicle.brakePadWear < 20 && vehicle.brakePadWear > 0) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "BRAKE PADS WORN!";
        aiData.alertLevel = 2;
        aiData.alertActive = true;
    }
    else if (vehicle.brakePadWear < 40) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "BRAKE PADS WEARING";
        aiData.alertLevel = 1;
        aiData.alertActive = true;
    }
    
    // Check 6: Tyre Pressure
    else if (vehicle.tyrePressureFL < 25 || vehicle.tyrePressureFR < 25 ||
             vehicle.tyrePressureRL < 25 || vehicle.tyrePressureRR < 25) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "LOW TYRE PRESSURE!";
        aiData.alertLevel = 2;
        aiData.alertActive = true;
    }
    else if (vehicle.tyrePressureFL < 28 || vehicle.tyrePressureFR < 28 ||
             vehicle.tyrePressureRL < 28 || vehicle.tyrePressureRR < 28) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "TYRE PRESSURE LOW";
        aiData.alertLevel = 1;
        aiData.alertActive = true;
    }
    
    // Check 7: Oil Quality Critical
    else if (vehicle.oilQuality < 20 && vehicle.oilQuality > 0) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "OIL QUALITY CRITICAL!";
        aiData.alertLevel = 2;
        aiData.alertActive = true;
    }
    else if (vehicle.oilQuality < 40) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "OIL QUALITY LOW";
        aiData.alertLevel = 1;
        aiData.alertActive = true;
    }
    
    // Check 8: TPMS Warning
    else if (vehicle.tpmsWarning) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "TPMS WARNING!";
        aiData.alertLevel = 2;
        aiData.alertActive = true;
    }
    
    // Check 9: Check Engine
    else if (vehicle.checkEngineLight) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "CHECK ENGINE LIGHT!";
        aiData.alertLevel = 2;
        aiData.alertActive = true;
    }
    
    // Check 10: Low Battery Voltage
    else if (vehicle.batteryVoltage < 12.0 && vehicle.batteryVoltage > 0) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "LOW BATTERY VOLTAGE";
        aiData.alertLevel = 1;
        aiData.alertActive = true;
    }
    else if (vehicle.batteryVoltage < 11.0) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "BATTERY VOLTAGE CRITICAL!";
        aiData.alertLevel = 2;
        aiData.alertActive = true;
    }
    
    // Check 11: Gear Mismatch
    else if (vehicle.speed > 100 && vehicle.rpm < 1500 && vehicle.speed > 0 && vehicle.rpm > 0) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "GEAR MISMATCH DETECTED";
        aiData.alertLevel = 1;
        aiData.alertActive = true;
    }
    
    // Check 12: Low Oil Pressure
    else if (vehicle.oilPressure < 10 && vehicle.oilPressure > 0) {
        aiData.anomalyDetected = true;
        aiData.alertMessage = "LOW OIL PRESSURE!";
        aiData.alertLevel = 2;
        aiData.alertActive = true;
    }
    
    // ---- Trigger Buzzer for Critical Alerts ----
    if (aiData.alertLevel == 2 && aiData.alertActive) {
        aiData.buzzerActive = true;
        aiData.alertStartTime = millis();
    } else {
        aiData.buzzerActive = false;
    }
    
    // ---- Print Alert (with 5 second cooldown) ----
    static unsigned long lastAlertPrint = 0;
    if (aiData.anomalyDetected && millis() - lastAlertPrint > 5000) {
        lastAlertPrint = millis();
        Serial.println("╔══════════════════════════════════════════════════════════════════╗");
        Serial.println("║                     🚨 AI ALERT DETECTED!                       ║");
        Serial.println("╠══════════════════════════════════════════════════════════════════╣");
        String level = (aiData.alertLevel == 2) ? "⚠️ CRITICAL" : "⚠️ WARNING";
        Serial.printf("║   Level: %-44s ║\n", level.c_str());
        Serial.printf("║   Message: %-44s ║\n", aiData.alertMessage.c_str());
        Serial.printf("║   Overall Health: %5.1f%%                                      ║\n", aiData.overallHealth);
        Serial.println("╚══════════════════════════════════════════════════════════════════╝");
        Serial.println();
    }
}

void storeHistoricalData() {
    history.engineHealth[history.index] = aiData.engineHealth;
    history.batteryHealth[history.index] = aiData.batteryHealth;
    history.brakeHealth[history.index] = aiData.brakeHealth;
    history.oilHealth[history.index] = aiData.oilHealth;

    history.rpm[history.index] = vehicle.rpm;
    history.speed[history.index] = vehicle.speed;
    history.coolantTemp[history.index] = vehicle.coolantTemp;
    history.timestamp[history.index] = millis();
    
    history.index = (history.index + 1) % HISTORY_SIZE;
    if (history.count < HISTORY_SIZE) {
        history.count++;
    }
}

// ============================================================
//                    UPDATE BUZZER
// ============================================================
void updateBuzzer() {
    if (aiData.buzzerActive) {
        unsigned long elapsed = millis() - aiData.alertStartTime;
        int position = elapsed % 400;
        
        if (position < 200) {
            digitalWrite(BUZZER_PIN, HIGH);
        } else {
            digitalWrite(BUZZER_PIN, LOW);
        }
        
        if (elapsed > 3000) {
            aiData.buzzerActive = false;
            digitalWrite(BUZZER_PIN, LOW);
        }
    } else {
        digitalWrite(BUZZER_PIN, LOW);
    }
}

    Serial.print(" | PS Fluid: "); Serial.print(100.0, 1); Serial.println("%");
    Serial.println("└────────────────────────────────────────────────────────────┘");

    Serial.println("\n┌─── 🔧 DIAGNOSTICS ───────────────────────────────────────┐");
    Serial.print("│ Check Engine: "); Serial.print(vehicle.checkEngineLight ? "⚠️ ON" : "✅ OK");
    Serial.print(" | Service: "); Serial.print(vehicle.serviceRequired ? "⚠️ REQ" : "✅ OK");
    Serial.print(" | ABS: "); Serial.println(vehicle.absWarning ? "⚠️ WARN" : "✅ OK");
    Serial.print("│ Traction Control: "); Serial.print(vehicle.tractionControlActive ? "🔄 ACTIVE" : "✅ OFF");
    Serial.print(" | ESP: "); Serial.print(vehicle.espActive ? "🔄 ACTIVE" : "✅ OFF");
    Serial.print(" | Brake Warning: "); Serial.println(vehicle.brakeWarning ? "⚠️ WARN" : "✅ OK");
    Serial.print("│ Oil Warning: "); Serial.print(vehicle.oilWarning ? "⚠️ WARN" : "✅ OK");
    Serial.print(" | Coolant Warning: "); Serial.print(vehicle.coolantWarning ? "⚠️ WARN" : "✅ OK");
    Serial.print(" | Battery Warning: "); Serial.println(vehicle.batteryWarning ? "⚠️ WARN" : "✅ OK");
    Serial.println("└────────────────────────────────────────────────────────────┘");
    
    // ─── GPS ───
    Serial.println("\n┌─── 📡 GPS ───────────────────────────────────────────────┐");
    Serial.print("│ Status: "); Serial.println(vehicle.latitude != 0 ? "✅ REAL GPS FIX" : "🔄 SIMULATED");
    Serial.print("│ Latitude: "); Serial.print(vehicle.latitude, 6);
    Serial.print(" | Longitude: "); Serial.println(vehicle.longitude, 6);
    Serial.print("│ Altitude: "); Serial.print(vehicle.altitude, 1); Serial.print("m");
    Serial.print(" | Satellites: "); Serial.print(vehicle.satellites);
    if (vehicle.speed > 0.5) {
        Serial.print(" | Speed: "); Serial.print(vehicle.speed, 1); Serial.print(" km/h");
        Serial.print(" | Course: "); Serial.print(0.0, 1); Serial.println("°");
    } else {
        Serial.println();
    }
    Serial.println("└────────────────────────────────────────────────────────────┘");
    
    // ─── CONTROLS & LED STATUS ───
    Serial.println("\n┌─── 🎮 CONTROLS & LED STATUS ───────────────────────────┐");
    Serial.print("│ Ignition: "); Serial.print(vehicle.engineRunning ? "🏁 ON" : "⏹ OFF");
    Serial.print(" | Mode: ");
    switch(vehicle.modeIndex) {
        case 0: Serial.print("ECO"); break;
        case 1: Serial.print("COMFORT"); break;
        case 2: Serial.print("SPORT"); break;
        case 3: Serial.print("SPORT+"); break;
        default: Serial.print("AUTO");
    }
    Serial.print(" | Gear: "); Serial.println(vehicle.gear);
    Serial.println("│");
    Serial.println("│ ─── POTENTIOMETERS ───");
    Serial.print("│ Throttle: "); Serial.print(vehicle.throttlePct); Serial.print("%");
    Serial.print("   | Brake: "); Serial.print(vehicle.brakePct); Serial.print("%");
    Serial.print("   | Steering: "); Serial.print(vehicle.steeringDeg); Serial.println("°");
    Serial.println("│");
    Serial.println("│ ─── LED STATUS ───");
    Serial.print("│ LED_GREEN (Engine): "); Serial.print(vehicle.engineRunning ? "🟢 ON" : "⚫ OFF");
    Serial.print("   | LED_WHITE (Check Eng): "); Serial.print(vehicle.checkEngineLight ? "⚪ ON" : "⚫ OFF");
    Serial.println();
    Serial.print("│ LED_RED (Brake): "); Serial.print((vehicle.brakePct > 60 || vehicle.brakeWarning) ? "🔴 ON" : "⚫ OFF");
    Serial.print("   | LED_BLUE (Headlight): "); Serial.print(vehicle.headlightOn ? "🔵 ON" : "⚫ OFF");
    Serial.println();
    Serial.println("│");
    Serial.println("│ ─── VIRTUAL BUTTONS ───");
    Serial.print("│ Headlight: "); Serial.print(vehicle.headlightOn ? "💡 ON" : "OFF");
    Serial.print("   | Horn: "); Serial.print(vehicle.hornOn ? "📢 ON" : "OFF");
    Serial.print("   | Wiper: "); Serial.println(vehicle.wiperOn ? "💧 ON" : "OFF");
    Serial.println("└────────────────────────────────────────────────────────────┘");
    
    // ─── CAN STATUS ───
    Serial.println("\n┌─── 📤 CAN STATUS ────────────────────────────────────────┐");
    Serial.print("│ Status: "); Serial.print(canStatus ? "✅ CONNECTED" : "❌ DISCONNECTED");
    Serial.print(" | Sent: "); Serial.print(0);
    Serial.print(" | Received: "); Serial.println(receiveCount);
    if (canStatus) {
        Serial.print("│ Last Send: "); Serial.print((millis() - lastHeartbeat) / 1000); Serial.print("s ago");
        Serial.print(" | Rate: "); Serial.print(receiveCount / ((millis() / 1000) > 0 ? (millis() / 1000) : 1)); Serial.println(" pkt/s");
    }
    Serial.println("└────────────────────────────────────────────────────────────┘");
    
    Serial.println("\n════════════════════════════════════════════════════════════\n");
