void setup() {

    Serial.println("WITH CAN TRANSMISSION");

    // ---- INITIALIZE HARDWARE PINS ----
    
    // LEDs
    pinMode(LED_PIN, OUTPUT);
    pinMode(LED_GREEN, OUTPUT);
    pinMode(LED_WHITE, OUTPUT);
    pinMode(LED_RED, OUTPUT);
    pinMode(LED_BLUE, OUTPUT);
    
    // Buttons (Input Pull-up)
    pinMode(BUTTON_IGNITION, INPUT_PULLUP);
    pinMode(BUTTON_MODE, INPUT_PULLUP);
    pinMode(BUTTON_FAILURE, INPUT_PULLUP);
    pinMode(BUTTON_CLEAR, INPUT_PULLUP);
    
    // MPU6050 (Optional - if connected, it will read, otherwise simulate)
    Wire.begin(I2C_SDA, I2C_SCL);
    mpu.initialize();
    mpu.setSleepEnabled(false);
    delay(100);
    
    mpu.getAcceleration(&ax, &ay, &az);
    if (ax != 0 || ay != 0 || az != 0) {
        Serial.println("MPU6050 CONNECTED! (Reading real data)");
        mpuOK = true;
    } else {
        Serial.println("MPU6050 NOT DETECTED! (Using simulated data)");
        mpuOK = false;
    }
    
    // GPS (Optional - if connected, it will read, otherwise simulate)
    GPS_SERIAL.begin(9600, SERIAL_8N1, GPS_RX, GPS_TX);
    Serial.println("GPS Init (Pins 16,17)");
    
    // CAN
    CAN.setPins(RX_GPIO_NUM, TX_GPIO_NUM);
    if (!CAN.begin(250E3)) {
        Serial.println("❌ Starting CAN failed!");
        canStatus = false;
    } else {
        Serial.println("✅ CAN Ready at 250KBPS!");
        canStatus = true;
    }
    
    Serial.println("\nVehicle Simulator Initialized");
    Serial.println("Hardware connected:");
    Serial.println("  - BUTTON_IGNITION (GPIO23)");
    Serial.println("  - BUTTON_MODE (GPIO19)");
    Serial.println("  - BUTTON_FAILURE (GPIO25)");
    Serial.println("  - BUTTON_CLEAR (GPIO26)");

    readMPU6050();
    
    // Read buttons from actual hardware

    // Read potentiometers from actual hardware
    readPots();
    
    // Update all simulated vehicle data based on inputs
    updateRealisticVehicleData();
    
    // Update virtual buttons (simulated automatic functions)
    updateVirtualButtons();
    
    // Update LEDs based on vehicle state
    updateLEDs();
    
    // Send all data via CAN every 2 seconds
    if (millis() - lastCANsend > 2000 && canStatus) {
        sendAllDataViaCAN();
        lastCANsend = millis();
    }
    
    // Receive CAN data
    if (canStatus) {
        receiveCANData();
    }
    
    // Print data every 2 seconds
    if (millis() - lastSend > 2000) {
        printRealisticData();
        lastSend = millis();
    }
    
    delay(10);
}

// ============================================================
//                    READ FUNCTIONS
// ============================================================
void readGPS() {
    // Read from actual GPS hardware if available
    while (GPS_SERIAL.available() > 0) {
        char c = GPS_SERIAL.read();
        gps.encode(c);
    }
    
    // If GPS has valid data, use it
    if (gps.location.isValid()) {
        gpsValid = true;
        vehicle.latitude = gps.location.lat();
        vehicle.longitude = gps.location.lng();
        vehicle.altitude = gps.altitude.meters();
        vehicle.satellites = gps.satellites.value();
        simulatedSpeed = gps.speed.kmph();
        simulatedCourse = gps.course.deg();
        simulatedSatellites = gps.satellites.value();
    } else {
        gpsValid = false;
    }
}

void readMPU6050() {
    if (mpuOK) {
        mpu.getAcceleration(&ax, &ay, &az);
        mpu.getRotation(&gx, &gy, &gz);
        tempC = mpu.getTemperature() / 340.0 + 36.53;
    }
}

void readPots() {
    throttlePct = map(analogRead(POT_THROTTLE), 0, 4095, 0, 100);
    brakePct = map(analogRead(POT_BRAKE), 0, 4095, 0, 100);
    steeringDeg = map(analogRead(POT_STEERING), 0, 4095, -45, 45);
    
    driveState.accelerationPedal = throttlePct / 100.0;
    driveState.brakePedal = brakePct / 100.0;
    driveState.steeringAngle = steeringDeg;
}

void readButtons() {
    unsigned long now = millis();
    
    bool reading = !digitalRead(BUTTON_IGNITION);
    if (reading != lastIgnition) {
        if (reading) {
            engineRunning = !engineRunning;
            if (engineRunning) {
                startEngine();
            } else {
                stopEngine();
            }
            pressCount++;
        }
        lastIgnition = reading;
    }
    
    reading = !digitalRead(BUTTON_MODE);
    if (reading != lastMode) {
        if (reading) {
            modeIndex = (modeIndex + 1) % 4;
            switch(modeIndex) {
                case 0: currentMode = "ECO"; break;
                case 1: currentMode = "COMFORT"; break;
                case 2: currentMode = "SPORT"; break;
                case 3: currentMode = "SPORT+"; break;
            }
            pressCount++;
            Serial.print("Drive Mode: ");
            Serial.println(currentMode);
        }
        lastMode = reading;
    }
    
    reading = !digitalRead(BUTTON_FAILURE);
    if (reading != lastFailure) {
        if (reading) {
            checkEngine = !checkEngine;
            pressCount++;
            Serial.print("Check Engine: ");
            Serial.println(checkEngine ? "ON" : "OFF");
            if (checkEngine) {
                vehicle.checkEngineLight = true;
                generateRandomFault();
            } else {
                clearFaults();
            }
        }
        lastFailure = reading;
    }
    
    reading = !digitalRead(BUTTON_CLEAR);
    if (reading != lastClear) {
        if (reading) {
            clearFaults();
            pressCount++;
            Serial.println("All Faults Cleared!");
        }
        lastClear = reading;
    }
}

void updateRealisticVehicleData() {
    static unsigned long lastUpdate = 0;
    static float dt = 0.01;
    
    if (millis() - lastUpdate < 100) return;
    dt = (millis() - lastUpdate) / 1000.0;
    lastUpdate = millis();
    
    // Convert pedal positions from hardware
    float throttleInput = driveState.accelerationPedal;
    float brakeInput = driveState.brakePedal;
    float steeringInput = driveState.steeringAngle / 45.0;
    
    // --- ENGINE SIMULATION ---
    if (engineRunning) {
        // Throttle response with realistic lag
        float targetThrottle = throttleInput * 0.85 + 0.15;
        driveState.throttleResponse += (targetThrottle - driveState.throttleResponse) * 2.5 * dt;
        
        // RPM calculation with inertia
        float loadFactor = 1.0 - (vehicle.speed / MAX_SPEED) * 0.3;
        float targetRPM = IDLE_RPM + (MAX_RPM - IDLE_RPM) * driveState.throttleResponse * loadFactor;
        
        // Engine braking when throttle released
        float engineBraking = 0.0;
        if (throttleInput < 0.05 && vehicle.speed > 5) {
            engineBraking = -150 * (vehicle.speed / 100.0);
        }
        
        // RPM with inertia
        float rpmInertia = 300.0;
        vehicle.rpm += (targetRPM - vehicle.rpm) / rpmInertia * 100.0 * dt;
        vehicle.rpm += engineBraking * dt;
        
        // RPM limits
        if (vehicle.rpm > REDLINE_RPM) {
            vehicle.rpm = REDLINE_RPM + (vehicle.rpm - REDLINE_RPM) * 0.5;
            Serial.println("⚠️ RPM LIMITER!");
        }
        if (vehicle.rpm < IDLE_RPM) vehicle.rpm = IDLE_RPM;
        if (vehicle.rpm > MAX_RPM) vehicle.rpm = MAX_RPM;
        
        // Engine load
        vehicle.engineLoad = 0.1 + (vehicle.rpm / MAX_RPM) * driveState.throttleResponse * 0.85;
        if (vehicle.engineLoad > 0.95) vehicle.engineLoad = 0.95;
        
        // Engine parameters
        vehicle.massAirFlow = vehicle.rpm / 60.0 * 0.5 * (0.5 + driveState.throttleResponse * 0.5);
        vehicle.fuelInjectorPulse = 2.0 + vehicle.engineLoad * 8.0;
        vehicle.ignitionTiming = 10.0 + (vehicle.rpm / 1000) * 5.0 - vehicle.engineLoad * 10.0;
        if (vehicle.ignitionTiming < 5) vehicle.ignitionTiming = 5;
        if (vehicle.ignitionTiming > 35) vehicle.ignitionTiming = 35;
        
        // Torque curve (peak at ~4000 RPM)
        float torqueCurve = 1.0 - 4.0 * pow((vehicle.rpm / MAX_RPM - 0.4), 2) + 0.3;
        if (torqueCurve < 0.3) torqueCurve = 0.3;
        vehicle.torque = 250.0 * torqueCurve * driveState.throttleResponse * 1.2;
        if (vehicle.torque < 0) vehicle.torque = 0;
        vehicle.horsepower = vehicle.torque * vehicle.rpm / 5252.0;
        
        // --- DRIVETRAIN AND SPEED (FIXED) ---
        float throttleFactor = driveState.accelerationPedal;
        float rpmFactor = (vehicle.rpm - IDLE_RPM) / (MAX_RPM - IDLE_RPM);
        if (rpmFactor < 0) rpmFactor = 0;
        if (rpmFactor > 1) rpmFactor = 1;
        
        // Simple but effective speed calculation
        float accelerationForce = throttleFactor * rpmFactor * 6000.0;
        float resistanceForce = 200.0 + (vehicle.speed * 1.5) + (vehicle.speed * vehicle.speed * 0.01);
        float netForce = accelerationForce - resistanceForce;
        float acceleration = netForce / 1500.0;
        
        // Apply braking
        if (brakeInput > 0.01) {
            float brakeForce = brakeInput * 5000.0;
            acceleration -= brakeForce / 1500.0;
            vehicle.brakeTemp += brakeInput * 10.0 * dt;
            vehicle.brakePadWear -= brakeInput * 0.001 * dt * 10;
            vehicle.brakeDiscWear -= brakeInput * 0.0005 * dt * 10;
            vehicle.brakePressure = brakeInput * 100.0;
        } else {
            vehicle.brakePressure = 0;
            if (vehicle.brakeTemp > 30) vehicle.brakeTemp -= 0.5 * dt * 10;
        }
        
        // Update speed
        vehicle.speed += acceleration * dt * 3.6;
        if (vehicle.speed < 0) vehicle.speed = 0;
        if (vehicle.speed > MAX_SPEED) vehicle.speed = MAX_SPEED;
        vehicle.acceleration = acceleration;
        
        // --- AUTOMATIC GEAR SELECTION ---
        if (vehicle.speed < 0.5) {
            vehicle.gear = 1;
        } else if (vehicle.speed < 20) {
            vehicle.gear = 1;
        } else if (vehicle.speed < 35) {
            vehicle.gear = 2;
        } else if (vehicle.speed < 55) {
            vehicle.gear = 3;
        } else if (vehicle.speed < 80) {
            vehicle.gear = 4;
        } else if (vehicle.speed < 110) {
            vehicle.gear = 5;
        } else {
            vehicle.gear = 6;
        }
        
        // Wheel speeds
        float wheelRPM = vehicle.speed / 3.6 / (2 * 3.14159 * 0.33) * 60;
        vehicle.driveShaftSpeed = wheelRPM * 3.5;
        
        // ABS and traction control
        if (brakeInput > 0.5 && vehicle.speed > 30) {
            vehicle.absWarning = true;
            if (brakeInput > 0.8) {
                vehicle.tractionControlActive = true;
            }
        } else {
            vehicle.absWarning = false;
            vehicle.tractionControlActive = false;
        }
        
        // ESP
        if (abs(steeringInput) > 0.5 && vehicle.speed > 50) {
            vehicle.espActive = true;
        } else {
            vehicle.espActive = false;
        }
        
         // --- REALISTIC TEMPERATURES ---
        float coolingEfficiency = 0.3 + (vehicle.speed / 150.0) * 0.7;
        if (coolingEfficiency > 1.0) coolingEfficiency = 1.0;
        if (coolingEfficiency < 0.3) coolingEfficiency = 0.3;
        
        float heatGeneration = (vehicle.rpm / MAX_RPM) * 0.4 + vehicle.engineLoad * 0.6;
        
        float targetCoolantTemp = 80.0 + (vehicle.engineLoad * 15.0) + (vehicle.rpm / MAX_RPM) * 5.0;
        if (targetCoolantTemp > 100.0) targetCoolantTemp = 100.0;
        
        vehicle.coolantTemp += (targetCoolantTemp - vehicle.coolantTemp) * 0.05 * dt;
        vehicle.coolantTemp += (heatGeneration * 5.0 * dt);
        vehicle.coolantTemp -= (coolingEfficiency * 1.5 * dt);
        
        if (vehicle.coolantTemp < 40.0) vehicle.coolantTemp = 40.0;
        if (vehicle.coolantTemp > 105.0) vehicle.coolantTemp = 105.0;
        
        float targetOilTemp = vehicle.coolantTemp + 5.0 + (vehicle.engineLoad * 10.0);
        if (targetOilTemp > 110.0) targetOilTemp = 110.0;
        
        vehicle.oilTemp += (targetOilTemp - vehicle.oilTemp) * 0.03 * dt;
        vehicle.oilTemp += (heatGeneration * 3.0 * dt);
        vehicle.oilTemp -= (coolingEfficiency * 0.5 * dt);
        
        if (vehicle.oilTemp < 40.0) vehicle.oilTemp = 40.0;
        if (vehicle.oilTemp > 110.0) vehicle.oilTemp = 110.0;
        
        vehicle.exhaustTemp = 200.0 + (vehicle.rpm / MAX_RPM) * 400.0 + (vehicle.engineLoad * 80.0);
        if (vehicle.exhaustTemp > 600.0) vehicle.exhaustTemp = 600.0;
        
        vehicle.catTemp = 300.0 + (vehicle.rpm / MAX_RPM) * 350.0;
        if (vehicle.catTemp > 650.0) vehicle.catTemp = 650.0;
        
        vehicle.oilPressure = 10.0 + (vehicle.rpm / MAX_RPM) * 40.0;
        if (vehicle.oilPressure < 5) vehicle.oilPressure = 5;
        
    } else {
        // Engine off
        vehicle.rpm = 0;
        vehicle.engineLoad = 0;
        vehicle.massAirFlow = 0;
        vehicle.fuelInjectorPulse = 0;
        vehicle.ignitionTiming = 0;
        vehicle.torque = 0;
        vehicle.horsepower = 0;
        
        // Coast to stop
        if (vehicle.speed > 0) {
            vehicle.speed -= 2.0 * dt;
            if (vehicle.speed < 0) vehicle.speed = 0;
        }
        vehicle.gear = 0;
        vehicle.driveShaftSpeed = 0;
        vehicle.tractionControlActive = false;
        vehicle.espActive = false;
        vehicle.absWarning = false;
        
        // Cool down
        if (vehicle.coolantTemp > 40.0) {
            vehicle.coolantTemp -= 0.5 * dt;
            if (vehicle.coolantTemp < 40.0) vehicle.coolantTemp = 40.0;
        }
        if (vehicle.oilTemp > 40.0) {
            vehicle.oilTemp -= 0.3 * dt;
            if (vehicle.oilTemp < 40.0) vehicle.oilTemp = 40.0;
        }
        if (vehicle.exhaustTemp > 30) vehicle.exhaustTemp -= 5.0 * dt;
        if (vehicle.catTemp > 30) vehicle.catTemp -= 8.0 * dt;
        if (vehicle.brakeTemp > 30) vehicle.brakeTemp -= 2.0 * dt;
    }
    
    // --- TYRE SIMULATION ---
    float tempFactorFL = (vehicle.tyreTempFL - 20) * 0.02;
    float tempFactorFR = (vehicle.tyreTempFR - 20) * 0.02;
    float tempFactorRL = (vehicle.tyreTempRL - 20) * 0.02;
    float tempFactorRR = (vehicle.tyreTempRR - 20) * 0.02;
    
    vehicle.tyrePressureFL = 35.0 + tempFactorFL * 5.0;
    vehicle.tyrePressureFR = 35.0 + tempFactorFR * 5.0;
    vehicle.tyrePressureRL = 35.0 + tempFactorRL * 5.0;
    vehicle.tyrePressureRR = 35.0 + tempFactorRR * 5.0;
    
    float speedHeat = vehicle.speed / 100.0;
    float steeringHeat = abs(steeringInput) * 0.3;
    float brakeHeat = brakeInput * 0.2;
    
    vehicle.tyreTempFL += (speedHeat * 5.0 + steeringHeat * 10.0 + brakeHeat * 5.0 - 0.5) * dt * 10;
    vehicle.tyreTempFR += (speedHeat * 5.0 + steeringHeat * 10.0 + brakeHeat * 5.0 - 0.5) * dt * 10;
    vehicle.tyreTempRL += (speedHeat * 4.0 + brakeHeat * 3.0 - 0.5) * dt * 10;
    vehicle.tyreTempRR += (speedHeat * 4.0 + brakeHeat * 3.0 - 0.5) * dt * 10;
    
    if (vehicle.tyreTempFL < 15) vehicle.tyreTempFL = 15;
    if (vehicle.tyreTempFR < 15) vehicle.tyreTempFR = 15;
    if (vehicle.tyreTempRL < 15) vehicle.tyreTempRL = 15;
    if (vehicle.tyreTempRR < 15) vehicle.tyreTempRR = 15;
    if (vehicle.tyreTempFL > 80) vehicle.tyreTempFL = 80;
    if (vehicle.tyreTempFR > 80) vehicle.tyreTempFR = 80;
    if (vehicle.tyreTempRL > 80) vehicle.tyreTempRL = 80;
    if (vehicle.tyreTempRR > 80) vehicle.tyreTempRR = 80;
    
    float wearRate = vehicle.speed / 200.0 + abs(steeringInput) * 0.2;
    if (vehicle.speed > 10) {
        vehicle.tyreWearFL -= wearRate * 0.001 * dt * 10;
        vehicle.tyreWearFR -= wearRate * 0.001 * dt * 10;
        vehicle.tyreWearRL -= wearRate * 0.0008 * dt * 10;
        vehicle.tyreWearRR -= wearRate * 0.0008 * dt * 10;
    }
    
    float tempGrip = 1.0 - abs(vehicle.tyreTempFL - 50) / 30.0;
    if (tempGrip < 0.5) tempGrip = 0.5;
    vehicle.tyreGripFL = tempGrip * (vehicle.tyreWearFL / 100.0);
    vehicle.tyreGripFR = tempGrip * (vehicle.tyreWearFR / 100.0);
    vehicle.tyreGripRL = tempGrip * (vehicle.tyreWearRL / 100.0);
    vehicle.tyreGripRR = tempGrip * (vehicle.tyreWearRR / 100.0);
    
    if (vehicle.tyrePressureFL < 30 || vehicle.tyrePressureFR < 30 ||
        vehicle.tyrePressureRL < 30 || vehicle.tyrePressureRR < 30) {
        vehicle.tpmsWarning = true;
    } else {
        vehicle.tpmsWarning = false;
    }
    
    // --- FUEL AND CONSUMPTION ---
    if (engineRunning) {
        float baseConsumption = 5.0 + (vehicle.rpm / MAX_RPM) * 15.0;
        float loadConsumption = vehicle.engineLoad * 8.0;
        vehicle.fuelConsumption = baseConsumption + loadConsumption;
        
        float fuelRate = vehicle.rpm / 1000.0 * 0.5 + vehicle.engineLoad * 2.0;
        vehicle.fuelLevel -= fuelRate * dt / 3600.0;
        
        vehicle.fuelPressure = 3.0 + vehicle.engineLoad * 2.0;
        vehicle.fuelTemp = vehicle.ambientTemp + vehicle.engineLoad * 20.0;
        
        float speedMS = vehicle.speed / 3.6;
        vehicle.distance += speedMS * dt / 1000.0;
        vehicle.tripDistance += speedMS * dt / 1000.0;
        vehicle.tripTime += dt / 3600.0;
        
        if (vehicle.speed > vehicle.maxSpeed) vehicle.maxSpeed = vehicle.speed;
        if (vehicle.tripTime > 0) {
            vehicle.avgSpeed = vehicle.tripDistance / vehicle.tripTime;
        }
        
    } else {
        vehicle.fuelConsumption = 0;
        vehicle.fuelPressure = 0;
    }
    
    if (vehicle.fuelLevel < 0) vehicle.fuelLevel = 0;
    if (vehicle.fuelLevel > FUEL_TANK_CAPACITY) vehicle.fuelLevel = FUEL_TANK_CAPACITY;
    
    // --- ELECTRICAL SYSTEM ---
    if (engineRunning) {
        float altEfficiency = 0.3 + (vehicle.rpm / MAX_RPM) * 0.7;
        vehicle.alternatorVoltage = 13.5 + altEfficiency * 1.0;
        vehicle.alternatorLoad = vehicle.engineLoad * 0.3;
        
        if (vehicle.alternatorVoltage > vehicle.batteryVoltage) {
            vehicle.batterySOC += (vehicle.alternatorVoltage - vehicle.batteryVoltage) * 0.5 * dt;
            vehicle.batteryHealth += 0.01 * dt;
        }
        
        vehicle.currentDraw = 30.0 + vehicle.engineLoad * 50.0;
        if (headlightOn) vehicle.currentDraw += 10.0;
        if (wiperOn) vehicle.currentDraw += 5.0;
        if (vehicle.acOn) vehicle.currentDraw += 15.0;
        
    } else {
        vehicle.batterySOC -= 0.01 * dt;
        vehicle.alternatorVoltage = 12.6;
        vehicle.alternatorLoad = 0;
        vehicle.currentDraw = 5.0;
    }
    
    vehicle.batteryVoltage = 12.0 + (vehicle.batterySOC / 100.0) * 2.0;
    if (vehicle.batteryVoltage < 11.5) {
        vehicle.batteryWarning = true;
    }
    
    if (vehicle.batterySOC < 0) vehicle.batterySOC = 0;
    if (vehicle.batterySOC > 100) vehicle.batterySOC = 100;
    if (vehicle.batteryHealth < 0) vehicle.batteryHealth = 0;
    if (vehicle.batteryHealth > 100) vehicle.batteryHealth = 100;
    if (vehicle.batteryVoltage < 10) vehicle.batteryVoltage = 10;
    if (vehicle.batteryVoltage > 15) vehicle.batteryVoltage = 15;
    
    // --- WEAR AND TEAR ---
    if (engineRunning) {
        float wearFactor = (vehicle.rpm / MAX_RPM) * 0.5 + vehicle.engineLoad * 0.5;
        vehicle.engineHealth -= wearFactor * 0.0001 * dt * 10;
        vehicle.engineHours += dt / 3600.0;
        vehicle.oilQuality -= (wearFactor * 0.0002 * dt * 10);
    }
    
    float shiftWear = abs(vehicle.rpm - 2000) / 2000.0 * 0.1;
    vehicle.transmissionHealth -= shiftWear * 0.0001 * dt * 10;
    
    if (vehicle.speed < 10 && driveState.accelerationPedal > 0.5) {
        vehicle.clutchWear -= 0.001 * dt * 10;
    }
    
    if (vehicle.engineHealth < 0) vehicle.engineHealth = 0;
    if (vehicle.oilQuality < 0) vehicle.oilQuality = 0;
    if (vehicle.transmissionHealth < 0) vehicle.transmissionHealth = 0;
    if (vehicle.clutchWear < 0) vehicle.clutchWear = 0;
    if (vehicle.brakePadWear < 0) vehicle.brakePadWear = 0;
    if (vehicle.brakeDiscWear < 0) vehicle.brakeDiscWear = 0;
    
    
    if (!checkEngine) {
        vehicle.checkEngineLight = 
            vehicle.engineHealth < 30 ||
            vehicle.oilQuality < 20 ||
            vehicle.transmissionHealth < 30 ||
            vehicle.brakePadWear < 10 ||
            vehicle.tyreWearFL < 15 ||
            vehicle.tyreWearFR < 15 ||
            vehicle.tyreWearRL < 15 ||
            vehicle.tyreWearRR < 15 ||
            vehicle.oilLevel < 10;
    }
    
    vehicle.serviceRequired = 
        vehicle.engineHealth < 50 ||
        vehicle.oilQuality < 40 ||
        vehicle.brakePadWear < 30 ||
        vehicle.transmissionHealth < 50 ||
        vehicle.tyreWearFL < 40 ||
        vehicle.tyreWearFR < 40 ||
        vehicle.tyreWearRL < 40 ||
        vehicle.tyreWearRR < 40 ||
        vehicle.oilLevel < 20;
    
    // --- ENVIRONMENTAL ---
    vehicle.ambientTemp = 22.0 + sin(millis() / 1000000.0 * 2 * 3.14159) * 5.0;
    vehicle.intakeAirTemp = vehicle.ambientTemp + (vehicle.rpm / MAX_RPM) * 15.0;
    
    // MPU data simulation (if hardware not connected)
    if (!mpuOK) {
        float accelX = vehicle.acceleration / 9.81 * 100;
        float accelY = steeringInput * 0.5 * (vehicle.speed / 100.0) * 100;
        float accelZ = -100;
        
        ax = accelX;
        ay = accelY;
        az = accelZ;
        gx = steeringInput * (vehicle.speed / 50.0) * 100;
        gy = -vehicle.acceleration / 9.81 * 200;
        gz = 0;
        tempC = vehicle.coolantTemp;
    }
    
    // GPS simulation (if hardware not connected)
    if (!gpsValid) {
        if (vehicle.speed > 0.5 && engineRunning) {
            float heading = atan2(steeringInput * 0.1, 1.0);
            float speedMS = vehicle.speed / 3.6;
            simulatedLat += speedMS * dt * sin(heading) / 111320.0;
            simulatedLng += speedMS * dt * cos(heading) / (111320.0 * cos(simulatedLat));
            simulatedSpeed = vehicle.speed;
            simulatedCourse = heading * 180 / 3.14159;
        } else {
            simulatedSpeed = 0;
        }
        
        vehicle.latitude = simulatedLat;
        vehicle.longitude = simulatedLng;
        vehicle.altitude = simulatedAltitude;
        vehicle.satellites = simulatedSatellites;
    }
}

void updateVirtualButtons() {
    unsigned long now = millis();
    
    if (now - lastHeadlightToggle > 10000) {
        lastHeadlightToggle = now;
        headlightOn = !headlightOn;
        Serial.print("💡 Headlight: ");
        Serial.println(headlightOn ? "ON" : "OFF");
    }
    
    if (now - lastHornToggle > 15000) {
        lastHornToggle = now;
        hornActive = true;
        hornStartTime = now;
        hornOn = true;
        Serial.println("📢 HORN!");
    }
    if (hornActive && now - hornStartTime > 1000) {
        hornActive = false;
        hornOn = false;
        Serial.println("Horn OFF");
    }
    
    if (now - lastWiperToggle > 8000) {
        lastWiperToggle = now;
        wiperOn = !wiperOn;
        Serial.print("💧 Wiper: ");
        Serial.println(wiperOn ? "ON" : "OFF");
    }
}

void updateLEDs() {
    digitalWrite(LED_GREEN, engineRunning);
    digitalWrite(LED_WHITE, checkEngine || vehicle.checkEngineLight);
    digitalWrite(LED_RED, driveState.brakePedal > 0.6 || vehicle.brakeWarning);
    digitalWrite(LED_BLUE, headlightOn);
}

void receiveCANData() {
    int packetSize = CAN.parsePacket();
    
    if (packetSize) {
        byte data[8];
        int i = 0;
        while (CAN.available() && i < 8) {
            data[i++] = CAN.read();
        }
        
        if (i >= 3) {
            receiveCount++;
            Serial.print("📥 CAN RX #");
            Serial.print(receiveCount);
            Serial.print(" | ID:0x");
            Serial.print(CAN.packetId(), HEX);
            Serial.print(" | From Node: ");
            Serial.print(data[0]);
            Serial.print(" | Msg: ");
            Serial.print(data[1]);
            if (i > 3) {
                Serial.print(" | Data: ");
                for(int j=3; j<i; j++) {
                    Serial.print(data[j], HEX);
                    Serial.print(" ");
                }
            }
            Serial.println();
        }
    }
}
